import argparse
import socket     
def run_client(host, port):
    """Connect to an echo server and exchange interactive messages."""
    try:
        with socket.create_connection((host, port)) as connection:
            with connection.makefile("rwb") as stream:
                print(f"Connected to {host}:{port}")
                print("Type a message and press Enter. Type 'exit' to close.")

                while True:
                    message = input("> ")
                    stream.write(f"{message}\n".encode("utf-8"))
                    stream.flush()

                    echo = stream.readline()
                    if not echo:
                        print("Server closed the connection.")
                        return 1

                    print(f"Echo from server: {echo.decode('utf-8').rstrip()}")
                    if message.strip().lower() == "exit":
                        return 0
    except ConnectionRefusedError:
        print(f"Connection refused: no server is running on {host}:{port}.")
        print(f"Run 'python echo_server.py {port}' in another terminal first.")
        print("Or run 'python run_echo.py' to start both automatically.")
        return 1
    except OSError as error:
        print(f"Client error: {error}")
        return 1
    except (EOFError, KeyboardInterrupt):
        print("\nClient stopped.")
        return 0


def main():
    parser = argparse.ArgumentParser(description="Connect to a TCP echo server.")
    parser.add_argument("host", nargs="?", default="localhost")
    parser.add_argument("port", nargs="?", type=int, default=5000)
    arguments = parser.parse_args()

    return run_client(arguments.host, arguments.port)


if __name__ == "__main__":
    raise SystemExit(main())

"""Concurrent TCP echo server."""

import argparse
import socketserver


class EchoRequestHandler(socketserver.StreamRequestHandler):
    """Echo each newline-delimited message from one client."""

    def handle(self):
        client_address = f"{self.client_address[0]}:{self.client_address[1]}"
        print(f"Client connected: {client_address}")

        for message in self.rfile:
            self.wfile.write(message)
            self.wfile.flush()

            if message.decode("utf-8").strip().lower() == "exit":
                break

        print(f"Client disconnected: {client_address}")


class ConcurrentEchoServer(socketserver.ThreadingTCPServer):
    allow_reuse_address = True
    daemon_threads = True


def create_server(port):
    """Create a concurrent echo server listening on the given port."""
    return ConcurrentEchoServer(("", port), EchoRequestHandler)


def main():
    parser = argparse.ArgumentParser(description="Run a concurrent TCP echo server.")
    parser.add_argument("port", nargs="?", type=int, default=5000)
    arguments = parser.parse_args()

    with create_server(arguments.port) as server:
        print(f"Concurrent echo server started on port {arguments.port}")
        print("Press Ctrl+C to stop the server.")
        try:
            server.serve_forever()
        except KeyboardInterrupt:
            print("\nServer stopped.")


if __name__ == "__main__":
    main()

"""Start a local concurrent echo server and interactive client together."""

import argparse
import threading

from echo_client import run_client
from echo_server import create_server


def main():
    parser = argparse.ArgumentParser(
        description="Run a local concurrent echo server and client."
    )
    parser.add_argument("port", nargs="?", type=int, default=5000)
    arguments = parser.parse_args()

    try:
        server = create_server(arguments.port)
    except OSError as error:
        print(f"Could not start the server on port {arguments.port}: {error}")
        return 1

    server_thread = threading.Thread(target=server.serve_forever, daemon=True)
    server_thread.start()
    print(f"Local concurrent echo server started on port {arguments.port}")

    try:
        return run_client("localhost", arguments.port)
    finally:
        server.shutdown()
        server.server_close()
        server_thread.join()
        print("Local server stopped.")


if __name__ == "__main__":
    raise SystemExit(main())

import socket
import threading


def handle_client(conn):

            break

    server.bind(('localhost', 5000))
    server.listen(5)
    print("server running")

import threading

a= 20
b= 50

condition = threading.Condition()
ready = False

def thread1():
    global ready

    with condition:
        print("Before Swapping:")
        print("a =", a)
        print("b =", b)

        ready = True
        condition.notify()

def thread2():
    global a, b

    with condition:
        while not ready:
            condition.wait()

            temp = a
            a= b
            b= temp

            print("\nAfter Swapping:")
            print("a =", a)
            print("b =", b)

t1 = threading.Thread(target=thread1)
t2 = threading.Thread(target=thread2)

t2.start()
t1.start()

t1.join()
t2.join()

import threading

counter = 0
lock = threading.Lock()
def increment():
    global counter
    for i in range(1000):
        with lock:
            counter += 1

t1 = threading.Thread(target=increment)
t2 = threading.Thread(target=increment)

t1.start()
t2.start()
t1.join()
t2.join()

print("Final counter value:", counter)

from xmlrpc.server import SimpleXMLRPCServer


class StatefulCounter:
    def __init__(self):
        self.count = 0

    def increment(self):
        self.count += 1
        return self.count


server = SimpleXMLRPCServer(("127.0.0.1", 8001), allow_none=True)
server.register_instance(StatefulCounter())
print("Stateful server running on port 8001")

try:
    server.serve_forever()
except KeyboardInterrupt:
    print("\nStateful server stopped")
    server.server_close()
